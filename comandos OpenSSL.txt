Laboratorio RSA

- openssl genrsa -out private.pem 2048
- openssl rsa -in private.pem -outform PEM -pubout -out public.pem
- cat private.pem
- echo "Mensaje secreto" > mensajesecreto.txt
- openssl pkeyutl -encrypt -pubin -inkey public.pem -in mensajesecreto.txt -out mensajesecretoencriptado.txt
- openssl pkeyutl -decrypt -inkey private.pem -in mensajesecretoencriptado.txt -out mensajerecuperado.txt 
- openssl dgst -sign private.pem -keyform PEM -sha256 -out mensajesecretosign.txt mensajesecreto.txt
- openssl dgst -verify public.pem -keyform PEM -sha256 -signature mensajesecretosign.txt mensajesecreto.txt
